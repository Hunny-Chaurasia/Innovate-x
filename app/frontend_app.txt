import React, {useEffect, useState} from 'react';
import ArchiveApp from './App.archive';
import {Institution, User, get, login, logout, register, session, setShellUser, validateSession} from './live/api';
import {ErrorMessage, Field, Form, Records, button, panel, text, useActivityRefresh} from './live/ui';
import {PublicPortfolio, route} from './live/student';

function Auth() {
  const [mode, setMode] = useState<'login' | 'student' | 'admin'>('login');
  const [institution, setInstitution] = useState<Institution>(); const [q, setQ] = useState('');
  return <main className="flex min-h-screen items-center justify-center bg-slate-950 p-5 font-['DM_Sans'] text-slate-100"><section className="w-full max-w-xl space-y-5"><header><p className="text-sm font-semibold text-violet-300">INNOVATEX · CAMPUS COLLABORATION</p><h1 className="mt-3 text-3xl font-semibold">Build something that matters.</h1><p className="mt-2 text-sm text-slate-400">Sign in with your real account. Expired or revoked sessions require a new sign-in.</p></header><div className="flex flex-wrap gap-2"><button className={button} onClick={() => setMode('login')}>Sign in</button><button className={button} onClick={() => setMode('student')}>Register student</button><button className={button} onClick={() => setMode('admin')}>First administrator</button></div>{mode === 'admin' && <p className="text-sm text-amber-300">Bootstrap is permitted only when no account exists. The server rejects subsequent administrator registrations. No default account is created.</p>}{mode === 'student' && <section className={panel}><h2 className="font-semibold">Select your institution</h2><input className="my-3 w-full rounded-lg border border-slate-600 bg-slate-950 p-3" value={q} onChange={event => setQ(event.target.value)} placeholder="Search institutions" aria-label="Institution search"/><Records<Institution> path={`/institutions?q=${encodeURIComponent(q)}`} publicRequest render={i => <button className={button} onClick={() => setInstitution(i)}>{institution?.id === i.id ? '✓ ' : ''}{i.name} · {i.kind}</button>}/><p className="mt-3 text-sm text-violet-300">{institution?.name || 'Choose an institution before registering.'}</p></section>}<section className={panel}><Form key={mode} label={mode === 'login' ? 'Sign in' : 'Create account'} onSubmit={async form => {
    if (mode === 'login') {await login(text(form, 'email'), text(form, 'password')); return;}
    if (mode === 'student' && !institution) throw new Error('Select your institution. If none are available, an administrator must create one first.');
    await register({email: text(form, 'email'), password: text(form, 'password'), display_name: text(form, 'name'), role: mode, ...(mode === 'student' && institution ? {institution_id: institution.id} : {})});
  }}>{mode !== 'login' && <Field name="name" label="Full name"/>}<Field name="email" label="Email" type="email"/><Field name="password" label="Password (12–128 characters)" type="password" minLength={12}/></Form></section><p className="text-xs text-slate-500">Only your bearer session is saved in this tab. Workflow records are fetched from the API.</p></section></main>;
}
function locationPath(): string {return window.location.hash.startsWith('#/') ? window.location.hash.slice(1).split('?')[0] : window.location.pathname;}
export default function App() {
  useActivityRefresh();
  const [path, setPath] = useState(locationPath()); const [user, setUser] = useState<User>(); const [loading, setLoading] = useState(true); const [error, setError] = useState<unknown>(); const [revision, setRevision] = useState(0);
  const publicSlug = path.startsWith('/u/') ? path.slice(3).split('/')[0] : '';
  useEffect(() => {const update = () => setPath(locationPath()); window.addEventListener('hashchange', update); window.addEventListener('popstate', update); return () => {window.removeEventListener('hashchange', update); window.removeEventListener('popstate', update);};}, []);
  useEffect(() => {const update = () => setRevision(n => n + 1); window.addEventListener('innovatex:session', update); window.addEventListener('focus', update); return () => {window.removeEventListener('innovatex:session', update); window.removeEventListener('focus', update);};}, []);
  useEffect(() => {
    let current = true;
    if (publicSlug) {setLoading(false); return;}
    const saved = session();
    if (!saved) {setShellUser(); setUser(undefined); setLoading(false); setError(undefined); return;}
    setError(undefined);
    const expiry = window.setTimeout(() => setRevision(n => n + 1), Math.max(0, Date.parse(saved.expires) - Date.now()));
    validateSession().then(value => {if (current) {setUser(value); setLoading(false);}}).catch(failure => {if (current) {setShellUser(); setUser(undefined); setError(failure); setLoading(false);}});
    return () => {current = false; window.clearTimeout(expiry);};
  }, [revision, publicSlug]);
  if (publicSlug) return <PublicPortfolio key={publicSlug} slug={publicSlug}/>;
  if (loading) return <main className="min-h-screen bg-slate-950 p-8 text-slate-300" role="status">Checking session…</main>;
  if (!user && path === '/') return <ArchiveApp/>;
  if (!user) return <>{error !== undefined && <div className="bg-slate-950 p-4"><ErrorMessage error={error}/><button className={button} onClick={() => setRevision(n => n + 1)}>Retry session check</button></div>}<Auth/></>;
  if (user.role === 'admin' || user.role === 'mentor') return <main className="min-h-screen bg-slate-950 p-8 font-['DM_Sans'] text-slate-100"><section className={panel}><h1 className="text-xl font-semibold">Account ready · {user.virtual_id}</h1><p className="my-3">Your {user.role} account is authenticated. A live workspace for this role is not available in this integration.</p><Form label="Sign out" onSubmit={logout}><span className="sr-only">Sign out</span></Form></section></main>;
  const prefix = `/${user.role}`;
  if (path !== prefix && !path.startsWith(`${prefix}/`)) return <main className="min-h-screen bg-slate-950 p-8 text-slate-100"><section className={panel}><h1 className="text-xl">Signed in as {user.display_name}</h1><p className="my-3">This account can open the {user.role} workspace, not another role's workspace.</p><a className={button} href={route(prefix)}>Continue to your workspace →</a><Form label="Sign out" onSubmit={logout}><span className="sr-only">Sign out</span></Form></section></main>;
  return <ArchiveApp/>;
}
