import React, {useState} from 'react';
import {CSR, Funding, Invitation, Page, Problem, Project, Reply, Review, User, Visit, Workspace, currentUser, domains, get, logout, request, safeLink, write} from './api';
import {Boundary, Chip, Field, Form, Records, Select, button, input, panel, text, useResource} from './ui';
import {ProofOfWork} from './student';

export function Invitations({user}: {user: User}) {
  return <section className="space-y-3"><h2 className="text-lg font-semibold">Collaboration invitations</h2><Records<Invitation> path="/requests" render={r => <article className={panel}><Chip>{r.kind} · {r.status}</Chip><p className="my-3">{r.reason}</p><p className="text-xs text-slate-400">Project {r.project_id}</p><p className="text-sm text-slate-400">{r.response_note}</p>{r.recipient_id === user.id && r.status === 'pending' && <Form label="Record invitation decision" onSubmit={async f => {await write(`/requests/${r.id}/respond`, {status: text(f, 'status'), note: text(f, 'note')});}}><Select name="status" label="Decision"><option value="accepted">Accept</option><option value="declined">Decline</option></Select><Field name="note" label="Response note" required={false}/></Form>}</article>}/></section>;
}
export function Summary({paths}: {paths: Record<string, string>}) {
  const resource = useResource<Record<string, number>>(JSON.stringify(paths), async () => Object.fromEntries(await Promise.all(Object.entries(paths).map(async ([label, path]) => [label, (await request<Page<{id: string}>>(`${path}${path.includes('?') ? '&' : '?'}limit=1`)).total]))));
  return <Boundary resource={resource}>{stats => <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">{Object.entries(stats).map(([label, count]) => <section key={label} className={panel}><p className="text-2xl font-semibold">{count}</p><p className="text-sm text-slate-400">{label}</p></section>)}</div>}</Boundary>;
}
export function ProjectHeader({project: p}: {project: Workspace}) {return <section className={panel}><Chip>{p.stage} · {p.team.status}</Chip><h2 className="my-3 text-xl font-semibold">{p.title}</h2><p className="whitespace-pre-wrap text-sm text-slate-300">{p.description}</p><progress className="mt-4 h-2 w-full accent-current" max={100} value={p.progress}/><p className="text-xs text-slate-400">{p.progress}% · {p.team.name}</p><div className="mt-4 divide-y divide-slate-700">{p.memberships.map(m => <div key={m.id} className="flex flex-wrap justify-between gap-2 py-2 text-sm"><span>{m.display_name} · <span className="font-mono">{m.virtual_id}</span>{m.user_id === p.team.leader_id ? ' · Leader' : ''}</span><Chip>{m.status}</Chip></div>)}</div><p className="mt-3 text-xs text-slate-400">Accepted mentor: {p.mentor_id || 'Awaiting acceptance'}</p></section>;}
function IndustryThread({review: r, user}: {review: Review; user: User}) {return <article className={panel}><Chip>{r.status}</Chip><h3 className="my-2 font-semibold">{r.summary}</h3><div className="space-y-2 text-sm text-slate-300"><p>Strengths: {r.strengths}</p><p>Flaws: {r.flaws}</p><p>Improvements: {r.improvements}</p></div><Records<Reply> path={`/reviews/${r.id}/replies`} render={reply => <div className="my-3 border-l border-amber-700 pl-3"><p className="text-xs text-amber-300">{reply.author_role} · {reply.action}</p><p className="text-sm">{reply.body}</p></div>}/>{r.reviewer_id === user.id && r.status !== 'resolved' && <Form label="Send review follow-up" onSubmit={async f => {await write(`/reviews/${r.id}/replies`, {body: text(f, 'body'), action: text(f, 'action')});}}><Field name="body" label="Follow-up and evidence assessment"/><Select name="action" label="Review decision"><option value="comment">Comment</option>{r.status === 'addressed' && <option value="request_changes">Request further changes</option>}<option value="resolve">Resolve review</option></Select></Form>}</article>;}
function FundingForm({id}: {id: string}) {return <section className={panel}><h3 className="mb-3 font-semibold">Record a completed transfer</h3><p className="mb-3 text-sm text-slate-400">This records an external transfer, not a payment. Funding is pending until the designated leader confirms receipt.</p><Form label="Submit transfer for confirmation" onSubmit={async f => {const proof = safeLink(text(f, 'proof_url')); if (!proof) throw new Error('Provide a durable HTTP(S) transfer evidence URL.'); await write(`/projects/${id}/funding`, {amount_lakh: text(f, 'amount_lakh'), txn_ref: text(f, 'txn_ref'), proof_url: proof, transfer_confirmed: f.get('confirmed') === 'on'});}}><label className="block text-sm">Amount (₹ lakh)<input className={input} name="amount_lakh" type="number" min="0.000001" step="0.000001" required/></label><Field name="txn_ref" label="Bank transaction reference"/><Field name="proof_url" label="Durable transfer proof URL" type="url"/><label className="block text-sm"><input name="confirmed" type="checkbox" required/> I confirm I have already transferred this amount.</label></Form><div className="mt-4"><Records<Funding> path={`/funding?project_id=${id}`} render={f => <div className="border-t border-slate-700 py-3"><Chip>{f.status}</Chip><p>₹{Number(f.amount_lakh).toFixed(2)} lakh · {f.txn_ref}</p><p className="text-sm text-slate-400">{f.decline_note}</p>{f.proof_url && safeLink(f.proof_url) && <a className="text-sm text-amber-300 underline" href={safeLink(f.proof_url)} target="_blank" rel="noopener noreferrer">Transfer evidence</a>}</div>}/></div></section>;}
function ReviewVisit({visit, user, close}: {visit: Visit; user: User; close: () => void}) {
  const id = visit.project.id;
  return <div className="space-y-5"><ProjectHeader project={visit.project}/><p className="text-sm text-amber-300">A structured review or follow-up is required before closing this review visit. Leaving the page does not close the audit visit; reopening resumes it.</p><ProofOfWork projectId={id}/><section className={panel}><h3 className="mb-3 font-semibold">Structured project review</h3><Form label="Submit structured review" onSubmit={async f => {await write(`/projects/${id}/reviews`, {summary: text(f, 'summary'), strengths: text(f, 'strengths'), flaws: text(f, 'flaws'), improvements: text(f, 'improvements')});}}><Field name="summary" label="Review summary" minLength={10}/><Field name="strengths" label="Strengths"/><Field name="flaws" label="Flaws / risks" minLength={10}/><Field name="improvements" label="Actionable improvements" minLength={10}/></Form></section><Records<Review> path={`/projects/${id}/reviews`} render={r => <IndustryThread review={r} user={user}/>}/><FundingForm id={id}/><Form label="Close review visit and return" onSubmit={async () => {await write(`/projects/${id}/views/close`, {view_id: visit.view_id}); close();}}><p className="text-sm text-slate-400">The server verifies that a review or follow-up was recorded during this visit.</p></Form></div>;
}
function Discovery({user}: {user: User}) {
  const [q, setQ] = useState(''); const [stage, setStage] = useState(''); const [visit, setVisit] = useState<Visit>();
  if (visit) return <ReviewVisit visit={visit} user={user} close={() => setVisit(undefined)}/>;
  return <section className="space-y-4"><input className={input} value={q} onChange={e => setQ(e.target.value)} aria-label="Search projects" placeholder="Search project titles"/><label className="block text-sm">Development stage<select className={input} value={stage} onChange={e => setStage(e.target.value)}><option value="">All stages</option>{['Ideation', 'Building', 'Prototype', 'Completed', 'Funded'].map(s => <option key={s}>{s}</option>)}</select></label><Records<Project> path={`/projects?q=${encodeURIComponent(q)}&stage=${stage}`} render={p => <article className={panel}><div className="flex justify-between gap-3"><h3 className="font-semibold">{p.title}</h3><Chip>{p.display_stage || p.stage}</Chip></div><p className="my-2 text-sm text-slate-400">{p.team_name} · {p.category} · {p.team_status}</p><progress className="my-3 h-2 w-full accent-amber-400" max={100} value={p.progress}/><Form label="Open audited review visit" onSubmit={async () => {setVisit(await write<Visit>(`/projects/${p.id}/views`));}}><p className="text-xs text-amber-300">{p.progress}% · Review required</p></Form></article>}/></section>;
}
function Publishing() {const [created, setCreated] = useState<Problem>(); return <section className={panel}><Form label="Publish problem statement" onSubmit={async f => {setCreated(await write<Problem>('/problems', {title: text(f, 'title'), description: text(f, 'description'), domain: text(f, 'domain'), tags: text(f, 'tags').split(',').map(t => t.trim()).filter(Boolean), csr: f.get('csr') === 'on', source: 'industry', status: 'published'}));}}><Field name="title" label="Problem title" minLength={3}/><label className="block text-sm">Full problem brief<textarea className={input} name="description" required minLength={10} rows={6}/></label><Select name="domain" label="Domain">{domains.map(d => <option key={d}>{d}</option>)}</Select><Field name="tags" label="Tags (comma separated, up to 20)" required={false}/><label className="block text-sm"><input name="csr" type="checkbox"/> CSR-aligned problem</label></Form>{created && <p className="mt-3 text-amber-300">Published: {created.title}</p>}</section>;}
function CSRMetrics() {const resource = useResource<CSR>('csr', () => get<CSR>('/csr/summary')); return <Boundary resource={resource}>{s => <><div className="grid grid-cols-2 gap-3 lg:grid-cols-4">{[['Projects funded', s.projects_funded], ['Confirmed ₹ lakh', Number(s.confirmed_amount_lakh).toFixed(2)], ['Awaiting ₹ lakh', Number(s.awaiting_amount_lakh).toFixed(2)], ['Students reached', s.students_reached]].map(([label, value]) => <section key={label} className={panel}><p className="text-2xl font-semibold">{value}</p><p className="text-sm text-slate-400">{label}</p></section>)}</div><p className="text-sm text-slate-400">CSR-linked projects only. Confirmed amounts require leader receipt confirmation. Patent and startup outcomes are not tracked.</p></>}</Boundary>;}
export function IndustryPage({page}: {page: string}) {const user = currentUser(); return <main className="w-full min-w-0 space-y-5 bg-slate-950 p-4 font-['DM_Sans'] text-slate-100 md:p-6"><header className="flex flex-wrap items-center justify-between gap-3"><h1 className="text-2xl font-semibold text-amber-300">{page}</h1><Form label="Log out" onSubmit={logout}><span className="sr-only">End session</span></Form></header>{user?.role !== 'industry' ? <p>Industry access required.</p> : <>{page === 'Post Problem' ? <Publishing/> : page === 'Discover Projects' ? <Discovery user={user}/> : page === 'CSR Impact' ? <CSRMetrics/> : page === 'Industry Reviews' ? <Records<Review> path="/reviews" render={r => <IndustryThread review={r} user={user}/>}/> : <><section className="rounded-xl border border-amber-800 bg-amber-950 p-5 text-amber-100"><p className="text-xs uppercase tracking-widest">Industry Virtual ID</p><h2 className="my-2 text-xl">{user.display_name}</h2><p className="font-mono">{user.virtual_id}</p><p className="text-sm">{user.organization}</p></section><Summary paths={{'Discoverable projects': '/projects', 'Open reviews': '/reviews?status=open', 'Pending invitations': '/requests?direction=incoming&status=pending', 'Transfers awaiting receipt': '/funding?status=awaiting_leader'}}/><CSRMetrics/><Invitations user={user}/><Discovery user={user}/></>}</>}</main>;}
