export function ProofOfWork({projectId, user, editable = false, slug}: {projectId: string; user?: User; editable?: boolean; slug?: string}) {
  const path = slug ? `/public/portfolios/${encodeURIComponent(slug)}/projects/${projectId}/proofs` : `/projects/${projectId}/proofs`;
  return <section className="space-y-4"><h3 className="text-lg font-semibold">Proof of Work</h3><p className="text-sm text-slate-400">Publish durable external image, repository, or video links with backend metadata. Binary video storage is not provided by this experience. Do not publish confidential evidence.</p><Records<Proof> path={path} publicRequest={Boolean(slug)} render={proof => <EvidenceCard proof={proof} user={user} editable={editable}/>}/>{editable && <div className={panel}><Form label="Add proof" reset onSubmit={form => {
    const attachments: Attachment[] = text(form, 'urls').split('\n').map(url => url.trim()).filter(Boolean).map(url => {
      const normalized = url.includes('://') ? url : `https://${url}`;
      if (!safeLink(normalized)) throw new Error('Use valid HTTP(S) links without embedded credentials.');
      return {kind: text(form, 'kind') as Attachment['kind'], url: normalized, name: text(form, 'name'), source: 'external'};
    });
    if (!attachments.length || attachments.length > 30) throw new Error('Add between 1 and 30 links.');
    return write(`/projects/${projectId}/proofs`, {title: text(form, 'title'), description: text(form, 'description'), attachments});
  }}><Field name="title" label="Evidence title" minLength={3}/><Field name="description" label="What did you build or validate?" required={false}/><Select name="kind" label="Evidence type"><option value="link">Repository / link</option><option value="image">External image</option><option value="video">External video</option></Select><Field name="name" label="Link label" required={false}/><label className="block text-sm text-slate-300">External URLs, one per line<textarea className={input} name="urls" required rows={3}/></label></Form></div>}</section>;
}
