import React, {useState} from 'react';
import {Change, Milestone, PanelFeedback, Problem, Project, Team, User, Workspace, currentUser, domains, get, logout, write} from './api';
import {Boundary, Chip, Field, Form, Records, Select, button, input, panel, text, useResource} from './ui';
import {ProofOfWork} from './student';
import {Invitations, ProjectHeader, Summary} from './industry';

function Briefs() {
  const [q, setQ] = useState(''); const [domain, setDomain] = useState(''); const [view, setView] = useState('all'); const [brief, setBrief] = useState<Problem>();
  return <section className="space-y-4"><input className={input} value={q} onChange={e => setQ(e.target.value)} aria-label="Search problem briefs" placeholder="Search problem briefs"/><div className="flex flex-wrap gap-2">{['all', 'new', 'shortlisted', 'csr'].map(v => <button key={v} className={button} aria-pressed={v === view} onClick={() => setView(v)}>{v}</button>)}</div><label className="block text-sm">Domain<select className={input} value={domain} onChange={e => setDomain(e.target.value)}><option value="">All domains</option>{domains.map(d => <option key={d}>{d}</option>)}</select></label>{brief && <article className={panel}><button className={button} onClick={() => setBrief(undefined)}>Close brief</button><h2 className="my-3 text-xl">{brief.title}</h2><p className="whitespace-pre-wrap text-sm">{brief.description}</p><p className="mt-3 text-xs text-emerald-300">{brief.tags.join(' · ')}</p></article>}<Records<Problem> path={`/problems?q=${encodeURIComponent(q)}&domain=${domain}&view=${view}`} render={p => <article className={panel}><Chip>{p.domain} · {p.csr ? 'CSR' : p.source} · {p.seen_at ? 'Read' : 'Unread'}</Chip><h3 className="my-3 font-semibold">{p.title}</h3><p className="mb-3 line-clamp-3 text-sm text-slate-300">{p.description}</p><div className="grid gap-3 md:grid-cols-2"><Form label="Read full brief and mark read" onSubmit={async () => {setBrief(await get<Problem>(`/problems/${p.id}`));}}><span className="text-xs text-slate-400">Reading is recorded for your faculty account.</span></Form><Form label={p.shortlisted_at ? 'Remove from shortlist' : 'Shortlist problem'} onSubmit={async () => {await write(`/problems/${p.id}/shortlist`, {shortlisted: !p.shortlisted_at}, 'PUT');}} key={p.shortlisted_at || 'not-shortlisted'}><span className="text-xs text-emerald-300">{p.shortlisted_at ? 'Shortlisted' : 'Not shortlisted'}</span></Form></div></article>}/></section>;
}
function Students({user}: {user: User}) {const [q, setQ] = useState(''); const [created, setCreated] = useState<User>(); return <section className="space-y-5"><section className={panel}><h2 className="mb-3 font-semibold">Register a real student account</h2><Form label="Create student account" reset onSubmit={async f => {setCreated(await write<User>('/users', {display_name: text(f, 'name'), email: text(f, 'email'), password: text(f, 'password'), role: 'student', institution_id: user.institution_id}));}}><Field name="name" label="Student full name"/><Field name="email" label="Student email" type="email"/><Field name="password" label="Initial password (12–128 characters)" type="password" minLength={12}/><p className="text-xs text-slate-400">Use an agreed initial password and share it securely with the student. No credentials are generated. The student can change it in Settings.</p></Form>{created && <div className="mt-4 rounded-lg border border-emerald-800 bg-emerald-950 p-4 text-emerald-100" role="status"><p>{created.display_name} · Account created</p><p className="font-mono">{created.virtual_id}</p></div>}</section><input className={input} value={q} onChange={e => setQ(e.target.value)} aria-label="Search institution students" placeholder="Search name or Virtual ID"/><Records<User> path={`/users?role=student&institution_id=${user.institution_id}&q=${encodeURIComponent(q)}`} render={s => <div className={panel}><p>{s.display_name}</p><p className="font-mono text-sm text-emerald-300">{s.virtual_id}</p></div>}/></section>;}
function Formation({user}: {user: User}) {
  const [students, setStudents] = useState<User[]>([]); const [mentor, setMentor] = useState<User>(); const [problem, setProblem] = useState<Problem>(); const [q, setQ] = useState(''); const [mq, setMq] = useState(''); const [pq, setPq] = useState(''); const [created, setCreated] = useState('');
  if (user.institution_kind !== 'School') return <p className="text-sm text-emerald-300">Faculty team formation is available only for same-school teams. College and cross-institution teams are student-formed.</p>;
  if (created) return <section className={panel}><p className="text-emerald-300">Project created. Students have been added; the selected mentor must accept before the school team becomes active.</p><p className="text-xs text-slate-400">{created}</p><button className={button} onClick={() => {setCreated(''); setStudents([]); setMentor(undefined); setProblem(undefined);}}>Form another team</button></section>;
  return <section className="space-y-4"><h2 className="text-lg font-semibold">Form a school team</h2><input className={input} value={q} onChange={e => setQ(e.target.value)} aria-label="Select students" placeholder="Search your school's students"/><Records<User> path={`/users?role=student&institution_id=${user.institution_id}&q=${encodeURIComponent(q)}`} render={s => <button className={button} onClick={() => setStudents(old => old.some(x => x.id === s.id) ? old.filter(x => x.id !== s.id) : [...old, s])}>{students.some(x => x.id === s.id) ? '✓ ' : '+ '}{s.display_name} · {s.virtual_id}</button>}/><p className="text-sm text-emerald-300">Selected: {students.map(s => s.display_name).join(', ') || 'Choose 2–50 students'}</p><input className={input} value={mq} onChange={e => setMq(e.target.value)} aria-label="Search mentor" placeholder="Search mentor name or Virtual ID"/><Records<User> path={`/users?q=${encodeURIComponent(mq)}`} render={m => ['mentor', 'faculty', 'industry'].includes(m.role) && m.id !== user.id ? <button className={button} onClick={() => setMentor(m)}>{mentor?.id === m.id ? '✓ ' : ''}{m.display_name} · {m.virtual_id} · {m.role}</button> : null}/><p className="text-sm">Mentor: {mentor?.display_name || 'Required'}</p><input className={input} value={pq} onChange={e => setPq(e.target.value)} aria-label="Select project problem" placeholder="Search published problems"/><Records<Problem> path={`/problems?q=${encodeURIComponent(pq)}`} render={p => p.status === 'published' ? <button className={button} onClick={() => setProblem(p)}>{problem?.id === p.id ? '✓ ' : ''}{p.title}</button> : null}/><p className="text-sm">Problem: {problem?.title || 'Optional'}</p>{problem && <button className={button} onClick={() => setProblem(undefined)}>Clear selected problem</button>}<section className={panel}><Form label="Create project and request mentor" onSubmit={async f => {const leader = text(f, 'leader'); if (students.length < 2 || students.length > 50 || !students.some(s => s.id === leader) || !mentor) throw new Error('Select 2–50 students, a leader from that selection, and a mentor.'); const p = await write<Project>('/projects', {name: text(f, 'name'), title: text(f, 'title'), description: text(f, 'description'), category: text(f, 'category'), leader_id: leader, student_ids: students.map(s => s.id), mentor_id: mentor.id, ...(problem ? {problem_id: problem.id} : {}), reason: text(f, 'reason')}); setCreated(p.id);}}><Field name="name" label="Team name"/><Field name="title" label="Project title" minLength={3}/><Field name="description" label="Problem and proposed solution" minLength={10}/><Select name="category" label="Domain">{domains.map(d => <option key={d}>{d}</option>)}</Select><Select name="leader" label="Student team leader"><option value="">Select leader</option>{students.map(s => <option key={s.id} value={s.id}>{s.display_name} · {s.virtual_id}</option>)}</Select><Field name="reason" label="Mentor invitation and team purpose" minLength={10}/><label className="block text-sm"><input type="checkbox" required/> Confirm selected students belong to my school.</label></Form></section></section>;
}
function ChangeRequests({user}: {user: User}) {return <Records<Change> path="/institution-changes" render={c => <article className={panel}><Chip>{c.status}</Chip><p className="my-3">{c.reason}</p><p className="text-xs text-slate-400">Student: {c.student_id}</p><p className="text-sm">{c.response_note}</p>{c.status === 'pending' && c.faculty_id === user.id && <Form label="Record institution decision" onSubmit={async f => {await write(`/institution-changes/${c.id}/respond`, {status: text(f, 'status'), note: text(f, 'note')});}}><Select name="status" label="Decision"><option value="approved">Approve</option><option value="declined">Decline</option></Select><Field name="note" label="Decision note"/><p className="text-xs text-slate-400">Approval requires the student's existing memberships to be resolved first.</p></Form>}</article>}/>;}
function PanelForm({projectId, milestone}: {projectId: string; milestone?: Milestone}) {return <Form label="Submit panel feedback" onSubmit={async f => {await write(`/projects/${projectId}/panel-feedback`, {body: text(f, 'body'), score: text(f, 'score') ? text(f, 'score') : null, decision: text(f, 'decision'), ...(milestone ? {milestone_id: milestone.id} : {})});}}><Field name="body" label="Feedback and rationale"/><label className="block text-sm">Score (0–100, optional)<input className={input} name="score" type="number" min="0" max="100" step="0.01"/></label><Select name="decision" label="Panel decision"><option value="feedback">Feedback only</option>{milestone?.status === 'submitted' && <><option value="approved">Approve milestone</option><option value="changes_requested">Request changes</option></>}</Select></Form>;}
function FacultyWorkspace({id, close}: {id: string; close: () => void}) {
  const resource = useResource<Workspace>(id, () => get<Workspace>(`/projects/${id}`));
  return <section className="space-y-5"><button className={button} onClick={close}>← Institution projects</button><Boundary resource={resource}>{p => <ProjectHeader project={p}/>}</Boundary><ProofOfWork projectId={id}/><section className={panel}><h2 className="mb-3 text-lg font-semibold">Milestones and status decisions</h2><Records<Milestone> path={`/projects/${id}/milestones`} render={m => <section className="space-y-3 border-b border-slate-700 py-3"><div className="overflow-hidden rounded-lg border border-slate-700"><table className="table-auto w-full text-left text-sm"><thead className="bg-slate-800 text-slate-300"><tr><th className="p-2">Milestone</th><th className="p-2">Due</th><th className="p-2">Status</th></tr></thead><tbody><tr><td className="p-2">{m.title}</td><td className="p-2">{m.due_at ? new Date(m.due_at).toLocaleDateString() : 'Not set'}</td><td className="p-2"><Chip>{m.status}</Chip></td></tr></tbody></table></div><p className="text-sm text-slate-400">{m.description}</p><PanelForm projectId={id} milestone={m}/></section>}/><h3 className="my-4 font-semibold">Create milestone</h3><Form label="Create milestone" onSubmit={async f => {await write(`/projects/${id}/milestones`, {title: text(f, 'title'), description: text(f, 'description'), due_at: text(f, 'due_at') ? new Date(text(f, 'due_at')).toISOString() : null});}}><Field name="title" label="Milestone title"/><Field name="description" label="Expected deliverables" required={false}/><Field name="due_at" label="Due date" type="date" required={false}/></Form></section><section className={panel}><h2 className="mb-3 text-lg font-semibold">Project review panel</h2><PanelForm projectId={id}/><div className="mt-4"><Records<PanelFeedback> path={`/projects/${id}/panel-feedback`} render={f => <article className="border-t border-slate-700 py-3"><Chip>{f.decision}</Chip><p className="my-2 text-sm">{f.body}</p><p className="text-xs text-emerald-300">Score: {f.score === null ? 'Not scored' : Number(f.score).toFixed(2)} · Milestone: {f.milestone_id || 'Project-wide'}</p></article>}/></div></section></section>;
}
function InstitutionProjects() {const [q, setQ] = useState(''); const [id, setId] = useState(''); if (id) return <FacultyWorkspace id={id} close={() => setId('')}/>; return <section className="space-y-4"><input className={input} value={q} onChange={e => setQ(e.target.value)} aria-label="Search institution projects" placeholder="Search projects for proof inspection and review"/><Records<Project> path={`/projects?q=${encodeURIComponent(q)}`} render={p => <article className={panel}><div className="flex justify-between gap-3"><h3 className="font-semibold">{p.title}</h3><Chip>{p.stage}</Chip></div><p className="my-3 text-sm text-slate-400">{p.team_name} · {p.team_status} · {p.progress}%</p><progress className="mb-3 h-2 w-full accent-emerald-400" value={p.progress} max={100}/><button className={button} onClick={() => setId(p.id)}>Inspect proof and review milestones →</button></article>}/></section>;}
export function FacultyPage({page}: {page: string}) {const user = currentUser(); return <main className="w-full min-w-0 space-y-5 bg-slate-950 p-4 font-['DM_Sans'] text-slate-100 md:p-6"><header className="flex flex-wrap items-center justify-between gap-3"><h1 className="text-2xl font-semibold text-emerald-300">{page}</h1><Form label="Log out" onSubmit={logout}><span className="sr-only">End session</span></Form></header>{user?.role !== 'faculty' ? <p>Faculty access required.</p> : page === 'Students' ? <Students user={user}/> : page === 'Change Requests' ? <ChangeRequests user={user}/> : page === 'Problem Statements' ? <Briefs/> : page === 'Review Panel' ? <InstitutionProjects/> : page === 'Faculty Actions' ? <Invitations user={user}/> : <><section className="rounded-xl border border-emerald-800 bg-emerald-950 p-5 text-emerald-100"><p className="text-xs uppercase tracking-widest">Faculty Virtual ID</p><h2 className="my-2 text-xl">{user.display_name}</h2><p className="font-mono">{user.virtual_id}</p><p className="text-sm">{user.institution_name}</p></section><Summary paths={{'Accessible projects': '/projects', 'Institution teams': '/teams', 'Institution students': `/users?role=student&institution_id=${user.institution_id}`, 'Pending change requests': '/institution-changes?status=pending'}}/><Invitations user={user}/><InstitutionProjects/><section className="space-y-3"><h2 className="font-semibold">Institution teams</h2><Records<Team> path="/teams" render={t => <div className={panel}><p>{t.name} <Chip>{t.status}</Chip></p><p className="text-xs text-slate-400">Leader: {t.leader_id}</p></div>}/></section><Formation user={user}/></>}</main>;}
