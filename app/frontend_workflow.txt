type Row = Record<string, unknown>;
type Args<K extends keyof typeof Contract> = typeof Contract[K] extends (...args: infer A) => unknown ? A : never;
type Returns<K extends keyof typeof Contract> = typeof Contract[K] extends (...args: never[]) => infer R ? R : never;
const row = (value: unknown): Row => value && typeof value === 'object' ? value as Row : {};
const str = (value: unknown): string => typeof value === 'string' ? value : '';
const id = (value: unknown): string => typeof value === 'string' ? value : str(row(value).id);
const list = (value: unknown): unknown[] => Array.isArray(value) ? value : [];
function person(user: User): Contract.Person {return {...user, name: user.display_name, virtualId: user.virtual_id, institution: user.institution_id || user.organization || '', institutionId: user.institution_id, avatar: user.avatar_url || '', avatarUrl: user.avatar_url || ''} as unknown as Contract.Person;}
export let CONTACTS = [] as unknown as typeof Contract.CONTACTS;
export let TEAM_DIRECTORY = {} as typeof Contract.TEAM_DIRECTORY;
export let CURRENT_STUDENT = {} as typeof Contract.CURRENT_STUDENT;

let data: Row = empty();
let pending = 0;
let error: unknown;
let revision = 0;
let generation = 0;
const listeners = new Set<() => void>();
function empty(): Row {return {fundings: [], funding: [], reviews: [], replies: [], proofs: [], problems: [], projects: [], requests: [], collabRequests: [], students: [], registeredStudents: [], profiles: {}, shareProfiles: {}, seenProblems: [], seenProblemIds: [], shortlistedProblems: [], shortlistedProblemIds: []};}
function emit(): void {revision++; listeners.forEach(fn => fn()); window.dispatchEvent(new CustomEvent('innovatex:workflow', {detail: {pending, error, revision}}));}
function subscribe(fn: () => void): () => void {listeners.add(fn); return () => {listeners.delete(fn);};}
function normalized(value: unknown): Row {const source = row(value); const output: Row = {...source}; for (const [key, value] of Object.entries(source)) output[key.replace(/_([a-z])/g, (_, c: string) => c.toUpperCase())] = value; return output;}
async function all<T>(path: string): Promise<T[]> {const result: T[] = []; for (let offset = 0; offset <= 100000; offset += 100) {const page = await request<Page<T>>(`${path}${path.includes('?') ? '&' : '?'}limit=100&offset=${offset}`); result.push(...page.items); if (offset + page.items.length >= page.total || !page.items.length) break;} return result;}
async function load(): Promise<void> {
  const ticket = ++generation;
  if (!session()) {data = empty(); CONTACTS = [] as unknown as typeof CONTACTS; TEAM_DIRECTORY = {} as typeof TEAM_DIRECTORY; CURRENT_STUDENT = {} as typeof CURRENT_STUDENT; CURRENT_FACULTY = {} as typeof CURRENT_FACULTY; CURRENT_INDUSTRY = {} as typeof CURRENT_INDUSTRY; error = undefined; emit(); return;}
  pending++; error = undefined; emit();
  try {
    const user = await get<User>('/auth/me');
    const [users, projects, problems, requests, reviews, fundings, teams] = await Promise.all([all<User>('/users'), all<Project>('/projects'), all<Problem>('/problems'), all<Row>('/requests'), user.role === 'student' || user.role === 'industry' ? all<Review>('/reviews') : Promise.resolve([]), ['student', 'industry', 'admin'].includes(user.role) ? all<Funding>('/funding') : Promise.resolve([]), all<Row>('/teams')]);
    const share = user.role === 'student' ? await get<Share | null>('/portfolio/share') : null;
    const proofs = user.role === 'student' ? (await Promise.all(projects.map(p => all<Proof>(`/projects/${p.id}/proofs`)))).flat() : [];
    if (ticket !== generation) return;
    CONTACTS = users.map(person) as unknown as typeof CONTACTS;
    const mapped = person(user);
    CURRENT_STUDENT = (user.role === 'student' ? mapped : {}) as typeof CURRENT_STUDENT;
    CURRENT_FACULTY = (user.role === 'faculty' ? mapped : {}) as typeof CURRENT_FACULTY;
    CURRENT_INDUSTRY = (user.role === 'industry' ? mapped : {}) as typeof CURRENT_INDUSTRY;
    TEAM_DIRECTORY = Object.fromEntries(teams.map(t => [str(t.name), {...normalized(t), leader: users.map(person).find(p => id(p) === t.leader_id), members: list(t.members).map(normalized)}])) as unknown as typeof TEAM_DIRECTORY;
    data = {...empty(), projects: projects.map(p => ({...normalized(p), team: p.team_name, members: list(row(TEAM_DIRECTORY)[p.team_name] && row(row(TEAM_DIRECTORY)[p.team_name]).members)})), problems: problems.map(normalized), requests: requests.map(normalized), collabRequests: requests.map(normalized), reviews: reviews.map(r => ({...normalized(r), replies: []})), fundings: fundings.map(normalized), funding: fundings.map(normalized), proofs: proofs.map(normalized), students: users.filter(u => u.role === 'student').map(person), registeredStudents: users.filter(u => u.role === 'student').map(person), profiles: share ? {[user.id]: normalized(share)} : {}, shareProfiles: share ? {[user.id]: normalized(share)} : {}};
  } catch (failure) {if (ticket === generation) error = failure;}
  finally {pending--; emit();}
}
window.addEventListener('innovatex:refresh', () => {void load();});
window.addEventListener('innovatex:session', () => {void load();});
window.addEventListener('focus', () => {void load();});
export function useWorkflow(): ReturnType<typeof Contract.useWorkflow> & {pending: boolean; error: unknown; refresh: () => void} {useSyncExternalStore(subscribe, () => revision); useEffect(() => {void load();}, []); return {...data, pending: pending > 0, error, refresh} as ReturnType<typeof Contract.useWorkflow> & {pending: boolean; error: unknown; refresh: () => void};}
export function useAllProblems(): ReturnType<typeof Contract.useAllProblems> {useWorkflow(); return data.problems as ReturnType<typeof Contract.useAllProblems>;}
async function mutate<T>(operation: () => Promise<T>): Promise<T> {pending++; error = undefined; emit(); try {const result = await operation(); refresh(); return result;} catch (failure) {error = failure; emit(); throw failure;} finally {pending--; emit();}}
