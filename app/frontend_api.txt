export function currentShellUser(role?: string): ShellUser {
  if (!identity || (role && identity.role !== role)) throw new Error('An authenticated matching shell identity is required.');
  const initials = identity.display_name.trim().split(/\s+/).slice(0, 2).map(part => part[0]).join('').toUpperCase();
  return {...identity, name: identity.display_name, displayName: identity.display_name, virtualId: identity.virtual_id, institution: identity.institution_name || identity.institution_id || identity.organization, institutionType: identity.institution_kind, organization: identity.organization, avatar: identity.avatar_url || initials, avatarUrl: identity.avatar_url || '', initials} as unknown as ShellUser;
}
export async function validateSession(): Promise<User> {
  const user = await get<User>('/auth/me');
  if (user.institution_id) {
    for (let offset = 0; offset <= 100000; offset += 100) {
      const page = await request<Page<Institution>>(`/institutions?limit=100&offset=${offset}`, {public: true});
      const institution = page.items.find(i => i.id === user.institution_id);
      if (institution) {user.institution_name = institution.name; user.institution_kind = institution.kind; break;}
      if (offset + page.items.length >= page.total || !page.items.length) break;
    }
  }
  setShellUser(user); return user;
}
const storageKey = 'innovatex.session.v1';
let memory: Session | null = null;
export function session(): Session | null {
  try {
    const raw: unknown = JSON.parse(sessionStorage.getItem(storageKey) || 'null');
    if (raw && typeof raw === 'object' && 'token' in raw && 'expires' in raw && typeof raw.token === 'string' && typeof raw.expires === 'string') memory = {token: raw.token, expires: raw.expires};
  } catch { /* Storage may be disabled; memory-only sessions still work. */ }
  if (memory && !(Date.parse(memory.expires) > Date.now())) { clearSession(); return null; }
  return memory;
}
export function clearSession(): void {
  memory = null;
  setShellUser();
  try { sessionStorage.removeItem(storageKey); } catch { /* Storage unavailable. */ }
  window.dispatchEvent(new Event('innovatex:session'));
}
export function saveSession(value: SessionResult): void {
  setShellUser(value.user);
  memory = { token: value.access_token, expires: value.expires_at };
  try { sessionStorage.setItem(storageKey, JSON.stringify(memory)); } catch { /* Memory fallback. */ }
  window.dispatchEvent(new Event('innovatex:session'));
}

export async function request<T>(path: string, options: { method?: 'GET' | 'POST' | 'PUT' | 'PATCH' | 'DELETE'; body?: unknown; public?: boolean } = {}): Promise<T> {
  if (!path.startsWith('/') || path.startsWith('//') || path.includes('..') || path.includes('\\')) throw new ApiError(0, 'invalid_path', 'Invalid API path');
  const method = options.method || 'GET';
  const publicAllowed = (method === 'GET' && (path.startsWith('/institutions') || path.startsWith('/public/portfolios/'))) || (method === 'POST' && ['/auth/login', '/auth/register'].includes(path));
  if (options.public && !publicAllowed) throw new ApiError(0, 'invalid_public_request', 'This operation requires a session');
  const auth = options.public ? null : session();
  if (!options.public && !auth) throw new ApiError(401, 'session_expired', 'Your session has expired. Sign in again.');
  const controller = new AbortController();
  const timeout = window.setTimeout(() => controller.abort(), 20000);
  try {
    const response = await fetch(new URL(path.slice(1), base), {
      method, credentials: 'omit', cache: 'no-store', redirect: 'error', signal: controller.signal,
      headers: { Accept: 'application/json', ...(options.body === undefined ? {} : {'Content-Type': 'application/json'}), ...(auth ? {Authorization: `Bearer ${auth.token}`} : {}) },
      body: options.body === undefined ? undefined : JSON.stringify(options.body),
    });
    const payload: unknown = await response.json().catch(() => null);
    if (!response.ok) {
      if (response.status === 401 && auth && memory?.token === auth.token) clearSession();
      const envelope = payload as {error?: {code?: string; message?: string; details?: ErrorDetail[]}; request_id?: string} | null;
      throw new ApiError(response.status, envelope?.error?.code || 'request_failed', envelope?.error?.message || `Request failed (${response.status})`, envelope?.error?.details || [], envelope?.request_id || '');
    }
    if (!payload || typeof payload !== 'object' || (!('data' in payload) && !('items' in payload))) throw new ApiError(0, 'invalid_response', 'The server returned an unexpected response.');
    if (method !== 'GET' && !options.public) refresh();
    return payload as T;
  } catch (error) {
    if (error instanceof ApiError) throw error;
    throw new ApiError(0, controller.signal.aborted ? 'timeout' : 'network_error', controller.signal.aborted ? 'Request timed out. Refresh before retrying a write; it may have completed.' : 'Unable to reach the service. Check your connection and retry.');
  } finally { window.clearTimeout(timeout); }
}
export async function get<T>(path: string, isPublic = false): Promise<T> { return (await request<Result<T>>(path, {public: isPublic})).data; }
export async function write<T = unknown>(path: string, body?: unknown, method: 'POST' | 'PUT' | 'PATCH' | 'DELETE' = 'POST'): Promise<T> { return (await request<Result<T>>(path, {method, body})).data; }
export async function login(email: string, password: string): Promise<User> {
  const value = (await request<Result<SessionResult>>('/auth/login', {method: 'POST', body: {email, password}, public: true})).data;
  saveSession(value); return value.user;
}

  const value = (await request<Result<SessionResult>>('/auth/register', {method: 'POST', body, public: true})).data;
  saveSession(value); return value.user;
}
