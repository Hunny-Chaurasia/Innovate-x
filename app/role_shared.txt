import type { ReactNode } from 'react';
import { getUser } from './api';
import { mutate, refresh, useRecords, type Proof } from './records';
import { Field, panel, input, button } from './auth';
import { Form, Empty, text } from './StudentExperience';
export function RoleChip({ role, children }: { role: 'faculty' | 'industry'; children: ReactNode }) { return <span className={role === 'faculty' ? 'w-fit rounded-full bg-emerald-500/15 text-emerald-300 px-2 py-1 text-[10px] font-semibold uppercase' : 'w-fit rounded-full bg-amber-500/15 text-amber-300 px-2 py-1 text-[10px] font-semibold uppercase'}>{children}</span>; }
export function RoleFrame({ role, children }: { role: 'faculty' | 'industry'; children: ReactNode }) { const data = useRecords(); return <div className="w-full min-w-0 text-[var(--foreground)] font-['DM_Sans']"><p className="text-xs text-[var(--muted-foreground)] mb-4">{role === 'faculty' ? 'Faculty workspace' : 'Industry workspace'} · Refreshes while active; not a live socket feed.</p>{data.error && <p role="alert" className="text-red-400 text-sm mb-4">{data.error}</p>}{data.ready ? children : <div className={panel}><p role="status">{data.loading ? 'Loading workspace…' : 'Workspace unavailable'}</p><button className={button} disabled={data.loading} onClick={() => void refresh()}>Retry</button></div>}</div>; }
export function Invitations({ role }: { role: 'faculty' | 'industry' }) { const data = useRecords(); const incoming = data.requests.filter(r => r.recipient_id === getUser()?.id); return <section><h2 className="font-semibold mb-3">Mentor invitations</h2>{!incoming.length && <Empty>No incoming invitations.</Empty>}<div className="flex flex-col gap-4">{incoming.map(r => <article className={panel} key={r.id}><RoleChip role={role}>{r.status}</RoleChip><h3 className="text-sm font-semibold mt-3">{data.projects.find(p => p.id === r.project_id)?.title || 'Project invitation'}</h3><p className="text-xs my-3 whitespace-pre-wrap">{r.reason}</p>{r.status === 'pending' && <Form label="Respond to invitation" action={async f => { await mutate(`/requests/${r.id}/respond`, 'POST', { status: text(f, 'status'), note: text(f, 'note') }); }}><Field label="Decision"><select name="status" className={input}><option value="accepted">Accept</option><option value="declined">Decline</option></select></Field><Field label="Note"><textarea name="note" className={input} maxLength={5000} /></Field></Form>}</article>)}</div></section>; }
export function Evidence({ proofs }: { proofs: Proof[] }) { return <div className="flex flex-col gap-3">{!proofs.length && <Empty>No evidence submitted.</Empty>}{proofs.map(p => <article key={p.id} className={panel}><h3 className="font-semibold text-sm">{p.title}</h3><p className="text-xs whitespace-pre-wrap my-2">{p.description}</p><p className="text-xs text-[var(--muted-foreground)]">{new Date(p.created_at).toLocaleString()}</p>{p.attachments.map((a, i) => <a key={a.id || i} href={a.url} target="_blank" rel="noopener noreferrer" className="block text-xs text-indigo-300 mt-2">{a.name || a.kind} ↗</a>)}</article>)}</div>; }
