import { useState } from 'react';
import { Link } from 'react-router-dom';
import { getUser, all } from './api';
import { mutate, useRecords, type Project, type Proof, type Review } from './records';
import { Field, Status, panel, input, button } from './auth';
import { Form, Heading, Empty, domains, text } from './StudentExperience';
import { RoleFrame, RoleChip, Invitations, Evidence } from './RoleShared';
interface Visit { view_id: string; review_required: boolean; project: Project }
function ReviewThread({ review: r }: { review: Review }) {
  return <article className={panel}><RoleChip role="industry">{r.status}</RoleChip>{[['Assessment', r.summary], ['Strengths', r.strengths], ['Flaws', r.flaws], ['Improvements', r.improvements]].map(([title, body]) => <div key={title} className="my-3"><h3 className="text-xs font-semibold">{title}</h3><p className="text-sm whitespace-pre-wrap text-[var(--muted-foreground)]">{body || 'Not provided'}</p></div>)}{r.replies.map(reply => <div key={reply.id} className="border-t border-[var(--border)] py-3 text-xs"><RoleChip role="industry">{reply.author_role}</RoleChip><p className="mt-2 whitespace-pre-wrap">{reply.body}</p></div>)}{r.status !== 'resolved' && <Form label="Send follow-up" reset action={async f => { await mutate(`/reviews/${r.id}/replies`, 'POST', { body: text(f, 'body'), action: text(f, 'action') }); }}><Field label="Follow-up"><textarea name="body" required maxLength={10000} className={input} /></Field><Field label="Action"><select name="action" className={input}><option value="comment">Comment</option>{r.status === 'addressed' && <option value="request_changes">Request further changes</option>}<option value="resolve">Resolve review</option></select></Field></Form>}</article>;
}
function VisitModal({ visit, close }: { visit: Visit; close: () => void }) {
  const data = useRecords(); const p = visit.project; const [proofs, setProofs] = useState<Proof[]>([]); const [loaded, setLoaded] = useState(false); const [tab, setTab] = useState('Review');
  const reviews = data.reviews.filter(r => r.project_id === p.id);
  return <div className="fixed inset-0 z-50 bg-black/70 p-3 md:p-8 overflow-y-auto" role="dialog" aria-modal="true" aria-label="Project review visit"><section className="mx-auto max-w-4xl rounded-xl border border-[var(--border)] bg-[var(--background)] p-5 md:p-7"><Heading title={p.title} subtitle={`${p.team?.name || ''} · ${p.category} · Review visit`} /><p className="text-sm whitespace-pre-wrap mb-4">{p.description}</p><progress value={p.progress} max={100} className="w-full h-2 accent-amber-500" /><p className="text-xs mt-2 text-[var(--muted-foreground)]">{p.stage} · {p.progress}% · {p.team?.status}</p><div className="flex flex-wrap gap-3 my-5">{['Review', 'Funding', 'Team & evidence'].map(t => <button key={t} className={button} onClick={() => setTab(t)}>{t}</button>)}</div>
    {tab === 'Review' && <div className="flex flex-col gap-5"><p className="text-xs text-amber-300">A structured review or follow-up is required before closing this visit. Leaving the page does not clear that requirement.</p><Form label="Submit structured review" reset action={async f => { await mutate(`/projects/${p.id}/reviews`, 'POST', { summary: text(f, 'summary'), strengths: text(f, 'strengths'), flaws: text(f, 'flaws'), improvements: text(f, 'improvements') }); }}>{[['summary', 'Overall assessment', true], ['strengths', 'Strengths (optional)', false], ['flaws', 'Flaws / gaps', true], ['improvements', 'Actionable improvements', true]].map(([name, label, required]) => <Field key={String(name)} label={String(label)}><textarea name={String(name)} required={Boolean(required)} minLength={required ? 10 : undefined} maxLength={10000} rows={3} className={input} /></Field>)}</Form>{reviews.map(r => <ReviewThread key={r.id} review={r} />)}</div>}
    {tab === 'Funding' && <div className="flex flex-col gap-5"><FundingHistory projectId={p.id} /><Form label="Record completed transfer" reset action={async f => { await mutate(`/projects/${p.id}/funding`, 'POST', { amount_lakh: text(f, 'amount'), txn_ref: text(f, 'reference'), proof_url: text(f, 'proof') || null, transfer_confirmed: f.get('transferred') === 'on' }); }}><Field label="Amount (INR lakhs, up to 6 decimal places)"><input name="amount" type="number" min="0.000001" step="0.000001" required className={input} /></Field><Field label="Transaction reference"><input name="reference" required maxLength={240} className={input} /></Field><Field label="Transfer evidence URL (optional)"><input name="proof" type="url" className={input} /></Field><label className="text-xs flex gap-2"><input name="transferred" required type="checkbox" />I confirm the transfer has actually been made.</label><p className="text-xs text-[var(--muted-foreground)]">This records an external transfer, not a payment. Only the designated leader can confirm receipt. Pending transfers block another submission.</p></Form></div>}
    {tab === 'Team & evidence' && <div className="flex flex-col gap-4">{p.memberships?.map(m => <div className="flex justify-between border-b border-[var(--border)] py-2 text-xs" key={m.id}><span>{m.display_name} · {m.virtual_id}</span><RoleChip role="industry">{m.user_id === p.team?.leader_id ? 'Leader' : m.status}</RoleChip></div>)}<p className="text-xs">Mentor: {data.users.find(u => u.id === p.mentor_id)?.display_name || 'Not assigned'}</p><Form label="Load project evidence" action={async () => { setProofs(await all<Proof>(`/projects/${p.id}/proofs`)); setLoaded(true); }}><p className="text-xs text-[var(--muted-foreground)]">Evidence is loaded from the project record.</p></Form>{loaded && <Evidence proofs={proofs} />}<Form label="Invite mentor" reset action={async f => { await mutate(`/projects/${p.id}/requests`, 'POST', { kind: 'mentor', recipient_id: text(f, 'mentor'), reason: text(f, 'reason') }); }}><Field label="Mentor"><select required name="mentor" className={input}><option value="">Select mentor</option>{data.users.filter(u => ['mentor', 'faculty', 'industry'].includes(u.role) && u.id !== getUser()?.id).map(u => <option key={u.id} value={u.id}>{u.display_name} · {u.virtual_id}</option>)}</select></Field><Field label="Invitation reason"><textarea name="reason" minLength={10} required className={input} /></Field><p className="text-xs text-[var(--muted-foreground)]">An existing mentor or pending mentor invitation prevents replacement. The recipient must accept.</p></Form></div>}
    <div className="border-t border-[var(--border)] mt-6 pt-4"><Form label="Close review visit" action={async () => { await mutate(`/projects/${p.id}/views/close`, 'POST', { view_id: visit.view_id }); close(); }}><span className="text-xs text-[var(--muted-foreground)]">The server checks that you submitted a review or follow-up during this visit.</span></Form></div></section></div>;
}
function FundingHistory({ projectId = '' }: { projectId?: string }) {
  const data = useRecords(); const rows = data.funding.filter(f => !projectId || f.project_id === projectId);
  return <div className="overflow-hidden rounded-xl border border-[var(--border)]"><div className="overflow-x-auto"><table className="table-auto w-full text-xs"><thead className="bg-[var(--card)] text-[var(--muted-foreground)]"><tr><th className="p-3 text-left">Project / reference</th><th className="p-3 text-right">INR lakh</th><th className="p-3 text-left">Leader confirmation</th></tr></thead><tbody>{rows.map(f => <tr key={f.id} className="border-t border-[var(--border)] even:bg-[var(--card)]"><td className="p-3">{data.projects.find(p => p.id === f.project_id)?.title}<p className="text-[var(--muted-foreground)]">{f.txn_ref}</p>{f.proof_url && <a href={f.proof_url} target="_blank" rel="noopener noreferrer" className="text-amber-300">Transfer evidence ↗</a>}</td><td className="p-3 text-right">{Number(f.amount_lakh).toFixed(6)}</td><td className="p-3"><RoleChip role="industry">{f.status.replaceAll('_', ' ')}</RoleChip><p className="mt-2 text-red-400">{f.decline_note}</p></td></tr>)}</tbody></table></div>{!rows.length && <Empty>No funding records from your account.</Empty>}</div>;
}
function Discovery() {
  const data = useRecords(); const [query, setQuery] = useState(''); const [stage, setStage] = useState(''); const [visit, setVisit] = useState<Visit | null>(null);
  const projects = data.projects.filter(p => p.title.toLowerCase().includes(query.toLowerCase()) && (!stage || p.display_stage === stage || p.stage === stage));
  return <><Heading title="Discover Projects" subtitle="Campus ideas, real evidence, and meaningful industry feedback." /><div className="flex gap-3 mb-5"><input className={input} value={query} onChange={e => setQuery(e.target.value)} aria-label="Search projects" placeholder="Search projects" /><select className={input} value={stage} onChange={e => setStage(e.target.value)} aria-label="Stage"><option value="">All stages</option>{['Ideation', 'Building', 'Prototype', 'Completed', 'Funded'].map(s => <option key={s}>{s}</option>)}</select></div><div className="grid md:grid-cols-2 gap-4">{projects.map(p => <article className={panel} key={p.id}><RoleChip role="industry">{p.display_stage || p.stage}</RoleChip><h2 className="font-semibold mt-3">{p.title}</h2><p className="text-xs text-[var(--muted-foreground)] my-3">{p.team_name} · {p.category}</p><progress max={100} value={p.progress} className="w-full h-1.5 accent-amber-500" /><div className="mt-4"><Form label="Open project & review" action={async () => { setVisit(await mutate<Visit>(`/projects/${p.id}/views`, 'POST')); }}><span className="text-xs text-[var(--muted-foreground)]">{p.progress}% · Structured review required</span></Form></div></article>)}</div>{!projects.length && <Empty>No matching projects.</Empty>}{visit && <VisitModal visit={visit} close={() => setVisit(null)} />}</>;
}
function Publish() {
  return <><Heading title="Publish a Problem" subtitle="Give campus teams a clear challenge to work on." /><div className={panel}><Form label="Publish problem statement" reset action={async f => { await mutate('/problems', 'POST', { title: text(f, 'title'), description: text(f, 'description'), domain: text(f, 'domain'), source: text(f, 'source'), tags: text(f, 'tags').split(',').map(t => t.trim()).filter(Boolean), csr: f.get('csr') === 'on', status: 'published' }); }}><Field label="Title"><input name="title" required minLength={3} maxLength={240} className={input} /></Field><Field label="Problem brief"><textarea name="description" required minLength={10} maxLength={30000} rows={6} className={input} /></Field><Field label="Domain"><select name="domain" className={input}>{domains.map(d => <option key={d}>{d}</option>)}</select></Field><Field label="Source"><select name="source" className={input}><option value="industry">Industry</option><option value="community">Community</option></select></Field><Field label="Tags (comma separated, up to 20)"><input name="tags" className={input} /></Field><label className="flex gap-2 text-sm"><input type="checkbox" name="csr" />CSR eligible challenge</label></Form></div></>;
}
function CSR() { const { csr } = useRecords(); return <><Heading title="CSR Impact" subtitle="Server-calculated impact from CSR-linked projects and leader-confirmed transfers." /><div className="grid grid-cols-2 lg:grid-cols-4 gap-4 mb-5">{[['Projects funded', csr?.projects_funded ?? 0], ['Confirmed · INR lakh', Number(csr?.confirmed_amount_lakh || 0).toFixed(2)], ['Awaiting · INR lakh', Number(csr?.awaiting_amount_lakh || 0).toFixed(2)], ['Students reached', csr?.students_reached ?? 0]].map(([label, value]) => <div key={label} className={panel}><p className="text-xs text-[var(--muted-foreground)]">{label}</p><strong className="text-2xl block mt-3">{value}</strong></div>)}</div><p className="text-xs text-[var(--muted-foreground)] mb-5">Patents and startups are not tracked. Awaiting transfers are not counted as confirmed impact. Funding history below includes all your transfers.</p><FundingHistory /></>; }
function Dashboard() { const data = useRecords(); return <><Heading title="Industry Dashboard" subtitle="Partner with campus teams. Review progress. Fund meaningful work." /><div className="grid md:grid-cols-3 gap-4 mb-5"><div className={panel}><RoleChip role="industry">Industry Virtual ID</RoleChip><h2 className="font-semibold mt-3">{getUser()?.display_name}</h2><p className="font-mono text-amber-300 mt-2">{getUser()?.virtual_id}</p><p className="text-xs mt-2">{getUser()?.organization}</p></div><div className={panel}><p className="text-xs">Discoverable projects</p><strong className="text-3xl block my-3">{data.projects.length}</strong><Link to="/industry/discover" className="text-amber-300 text-xs">Open discovery →</Link></div><div className={panel}><p className="text-xs">Reviews awaiting follow-up</p><strong className="text-3xl block my-3">{data.reviews.filter(r => r.status === 'addressed').length}</strong><Link to="/industry/reviews" className="text-amber-300 text-xs">Review inbox →</Link></div></div><div className="flex gap-4 text-sm mb-5"><Link to="/industry/post">Publish problem →</Link><Link to="/industry/csr">CSR impact →</Link></div><h2 className="font-semibold mb-3">Your published challenges</h2><div className="grid md:grid-cols-2 gap-4 mb-5">{data.problems.filter(p => p.publisher_id === getUser()?.id).map(p => <article className={panel} key={p.id}><RoleChip role="industry">{p.domain}{p.csr ? ' · CSR' : ''}</RoleChip><h3 className="font-semibold mt-3">{p.title}</h3><p className="text-xs mt-2">{p.description}</p></article>)}</div><Invitations role="industry" /><h2 className="font-semibold my-4">Funding confirmation status</h2><FundingHistory /><p className="text-xs text-[var(--muted-foreground)] mt-5">Workshop scheduling is not available: no workshop API is connected.</p></>; }
export default function IndustryExperience({ page }: { page: string }) { const data = useRecords(); return <RoleFrame role="industry"><Status />{page === 'discover' ? <Discovery /> : page === 'post' ? <Publish /> : page === 'csr' ? <CSR /> : page === 'reviews' ? <><Heading title="Review Inbox" subtitle="Follow up on changes and resolve your structured reviews." />{!data.reviews.length && <Empty>No reviews yet. Open a project in discovery to start.</Empty>}<div className="flex flex-col gap-4">{data.reviews.map(r => <section key={r.id}><h2 className="font-semibold mb-2">{data.projects.find(p => p.id === r.project_id)?.title}</h2><ReviewThread review={r} /></section>)}</div></> : <Dashboard />}</RoleFrame>; }
